\section{Executive Summary}
The Soliva project is an e-commerce platform built with a React frontend and a
Node.js/Express backend. The repository contains the source code for a
public-facing storefront and an administrative dashboard. The codebase
implements user authentication, product catalog viewing, shopping cart
functionality, payment processing via Razorpay, and data persistence using
MongoDB.

% =========================================================
% 4. PROJECT OVERVIEW
% =========================================================
\section{Project Overview}
\begin{itemize}[leftmargin=*]
    \item \textbf{Application Purpose:} An online retail application allowing
          customers to purchase items and administrators to manage store data.
    \item \textbf{Target Users:} General shoppers and platform administrators.
    \item \textbf{Business Workflow:} Users browse products, add items to a
          cart, and execute payments through Razorpay. Administrators access a
          separate interface to manage inventory, track orders, and view
          transaction data.
    \item \textbf{Overall Functionality:} E-commerce storefront with product
          listing, cart, checkout, payment gateway integration, and a
          role-restricted admin panel.
\end{itemize}

% 7. OVERALL SYSTEM ARCHITECTURE
% =========================================================
\section{Overall System Architecture}
\subsection*{Request Lifecycle \& Flow}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Frontend Flow:} The Vite-built React application handles
          client-side routing via Tanstack Router and state via Zustand.
          API requests are managed by Tanstack Query.
    \item \textbf{API Flow:} The Express backend receives requests on port
          5000, applies middleware (Helmet, CORS, Rate Limiting), and maps
          them to controllers in the \texttt{routes} directory.
    \item \textbf{Authentication Flow:} Controllers require JWT validation
          middleware. The \texttt{/api/v1/admin} routes require an additional
          role check (\texttt{authorizeRoles('admin')}).
    \item \textbf{Database Flow:} Controllers interact with MongoDB using
          Mongoose schemas.
    \item \textbf{Payment Flow:} The backend calls Razorpay APIs to generate
          order IDs, and a webhook/verification endpoint validates Razorpay
          signatures to finalize orders.
\end{enumerate}

\section{Database Documentation}
\subsection*{Collections \& Schemas}
\begin{itemize}[leftmargin=*]
    \item \textbf{Users:} \texttt{name}, \texttt{email}, \texttt{password} (hashed), \texttt{role}.
    \item \textbf{UserProfiles:} Links to \texttt{Users}, contains extended data.
    \item \textbf{Products:} \texttt{name}, \texttt{description}, \texttt{price}, \texttt{stock}, \texttt{images}.
    \item \textbf{Orders:} Links to \texttt{Users} and \texttt{Products}, tracks
          \texttt{paymentStatus} and \texttt{orderStatus}.
    \item \textbf{Carts:} Temporary cart items array linking to \texttt{Products}.
    \item \textbf{Addresses:} \texttt{street}, \texttt{city}, \texttt{state}, \texttt{zipCode}.
    \item \textbf{AuthAuditLogs:} Tracks \texttt{ipAddress}, \texttt{action}, \texttt{status}.
    \item \textbf{PendingRegistrations:} Temporary holding for unverified users containing OTPs.
    \item \textbf{SupportTickets:} User-submitted inquiries.
\end{itemize}

\noindent\textbf{Implementation Details:} Schemas use Mongoose validation
constraints. No implementation secrets are stored directly in schema defaults.

% =========================================================
% 13. BUSINESS LOGIC
% =========================================================
\section{Business Logic}
\begin{itemize}[leftmargin=*]
    \item \textbf{Registration Flow:} User submits data $\rightarrow$ record
          created in \texttt{PendingRegistrations} with an OTP $\rightarrow$
          OTP verified $\rightarrow$ user moved to \texttt{Users} collection.
    \item \textbf{Order Processing:} User submits cart $\rightarrow$ backend
          calculates total against database prices (ignoring
          frontend-submitted prices) $\rightarrow$ generates Razorpay Order ID.
    \item \textbf{Payment Verification:} Frontend completes Razorpay modal
          $\rightarrow$ sends \texttt{razorpay\_payment\_id} and
          \texttt{razorpay\_signature} to backend $\rightarrow$ backend
          validates signature $\rightarrow$ modifies order status to Paid
          $\rightarrow$ BullMQ queue worker generates PDF invoice and sends
          email.
\end{itemize}

% =========================================================
% 14. SECURITY IMPLEMENTATION
% =========================================================
\section{Security Implementation}
\begin{itemize}[leftmargin=*]
    \item \textbf{JWT:} Stored in HTTP-only cookies (configured in
          \texttt{utils/jwtToken.js}) or passed via headers.
    \item \textbf{Password Hashing:} Implemented using \texttt{bcryptjs}.
    \item \textbf{Helmet:} Express middleware that sets HTTP response headers.
    \item \textbf{Rate Limiting:} \texttt{express-rate-limit} configured
          specifically for \texttt{/login} and \texttt{/forgot-password}
          routes to prevent brute force and SMS/Email bombing, and broadly
          on \texttt{/api}.
    \item \textbf{Sanitization:} \texttt{express-mongo-sanitize} removes keys
          starting with \texttt{\$} to prevent injection.
    \item \textbf{XSS Protection:} \texttt{xss-clean} strips script tags from
          request bodies.
    \item \textbf{CORS:} Allowed origins are strictly read from the
          \texttt{FRONTEND\_URL} environment variable.
\end{itemize}

          exceptions and formats standard JSON responses.

          bodies and schemas before executing controller logic.
